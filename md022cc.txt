           EXEC SQL
                DECLARE csr-cc-decline CURSOR WITH HOLD FOR
                SELECT
                       MHH.FK_ACL_CLUB_C
                     , MHH.MBRSHP_N
                     , MI.FK_CUST_SYS_ID_N
                  FROM
                       MBRSHP_HOUSEHOLD   MHH
                     , MBRSHP_NEXT_EVENT  NE
                     , MBR_INFO           MI
                  WHERE MHH.FK_ACL_CLUB_C  = MI.FK_MHH_ACL_CLUB_C
                          AND MHH.MBRSHP_N       = MI.FK_MHH_MBRSHP_N
                          AND MHH.FK_ACL_CLUB_C  = NE.FK_MHH_ACL_CLUB_C
                          AND MHH.MBRSHP_N       = NE.FK_MHH_MBRSHP_N
                          AND NE.NEXT_EVENT_D    = :WS-TARGET-DATE
                          AND MI.FK_MRO_ROLE_C   = '00'
                          AND MI.EXP_D           = '12/31/9999'
                          AND FK_MBR_BILL_EVENT  = '75'
                          AND MHH.FK_ACL_CLUB_C  <> '065'
               ORDER BY
                        MHH.FK_ACL_CLUB_C
                      , MHH.MBRSHP_N
              FOR FETCH ONLY
              WITH UR
           END-EXEC.
