           EXEC SQL
                DECLARE CUSTID CURSOR WITH HOLD FOR
                SELECT
                       MHH.FK_ACL_CLUB_C
                     , MHH.MBRSHP_N
                     , MI.FK_CUST_SYS_ID_N
                  FROM
                       MBRSHP_HOUSEHOLD MHH
                     , MBR_PRD_DTL MPD
                     , MBR_INFO MI
                  WHERE MHH.CUR_TERM_EXP_D < MHH.REN_TERM_EXP_D
                    AND MHH.FK_BILL_PLAN_TYP_C = 'AM'
                    AND MHH.FK_ACL_CLUB_C  = MI.FK_MHH_ACL_CLUB_C
                    AND MHH.MBRSHP_N       = MI.FK_MHH_MBRSHP_N
                    AND MI.FK_MRO_ROLE_C   = '00'
                    AND MI.EXP_D           = '12/31/9999'
                    AND MHH.FK_ACL_CLUB_C  = MPD.FK_MHH_ACL_CLUB_C
                    AND MHH.MBRSHP_N       = MPD.FK_MHH_MBRSHP_N
                    AND MPD.FK_MPR_MRO_ROLE_C = '00'
                    AND MPD.FK_MPR_PRD_C   = '01'
                    AND MPD.EXP_D          = '12/31/9999'
                    AND MPD.PAID_C         = 'N'
                    AND MPD.EFF_D          = :WS-TARGET-DATE
              ORDER BY
                       MHH.FK_ACL_CLUB_C
                     , MHH.MBRSHP_N
              FOR FETCH ONLY
              WITH UR
           END-EXEC.

           PERFORM 1015-OPEN-CURSOR THRU 1015-X.

           PERFORM 1020-FETCH THRU 1020-X.

           EVALUATE SQLCODE

                    ADD 1 TO WS-INPUT-COUNT

                      TO MD900-PARAGRAPH-NAME

                      TO MD900-TABLE-CURSOR-NAME
                    MOVE 'CURSOR FETCH ERROR'

                    PERFORM 9992-CLOSE-CURSOR THRU 9992-X
                    PERFORM 9999-ABEND THRU 9999-X

       1005-GET-PROC-DATE.

                                    + 13 DAYS,ISO)
           END-EXEC.

               WHEN MD903-SQL-OK
                    CONTINUE

                    MOVE '1005-GET-PROC-DATE'
                      TO MD900-PARAGRAPH-NAME

                    MOVE 'ERROR CALCULATING DATE'

             MOVE SPACES        TO WS-OUTREC
             MOVE ','           TO MD021-COMMA-01
                                 , MD021-COMMA-02
                                 , MD021-COMMA-03
                                 , MD021-COMMA-04
                                 , MD021-COMMA-05

             PERFORM 2015-CALL-MD999CK THRU 2015-X
             MOVE MD999CK-CHECK-DIGIT   TO MD021-MBR-NUM-16
             MOVE FK-CUST-SYS-ID-N OF MBR-INFO
               TO MD021-CUST-ID
             MOVE 'en-US'               TO MD021-LOCALS
             PERFORM 2001-CHECK-PIT     THRU 2001-X
             IF MEMBER-HAS-PIT-NO
                PERFORM 2010-CALL-MC501BCD THRU 2010-X
                PERFORM 2020-WRITE-OUTFILE THRU 2020-X
             ELSE
                ADD 1 TO WS-SKIPPED-COUNT
             END-IF
             PERFORM 1020-FETCH THRU 1020-X

             EVALUATE SQLCODE
                 WHEN MD903-SQL-OK
                      ADD 1 TO WS-INPUT-COUNT
                 WHEN 100
                      SET NO-MORE-TO-FETCH TO TRUE
                 WHEN OTHER
                      MOVE '1020-FETCH'
                        TO MD900-PARAGRAPH-NAME
                      SET MD900-DB2-FUNCTION-FETCH TO TRUE
                      MOVE 'CUSTID'
                        TO MD900-TABLE-CURSOR-NAME
                      MOVE 'CURSOR FETCH ERROR'
                        TO MD900-ERROR-MSG-1
                      PERFORM 9994-FILL-ABEND-DATA THRU 9994-X
                      PERFORM 9992-CLOSE-CURSOR THRU 9992-X
                      PERFORM 9999-ABEND THRU 9999-X
             END-EVALUATE
           END-PERFORM.
       2000-X.  EXIT.

       2001-CHECK-PIT.

           INITIALIZE FN435BT-COMMAREA
                    , WS-PIT-TOTAL.

           MOVE 'MD021EX '                 TO FIN435-I-PGM-NAME.
           SET  FIN435-MEMBERSHIP          TO TRUE.
           MOVE 'BATCH '                   TO FIN435-I-EMPLOYEE-NUM.
           MOVE FK-ACL-CLUB-C              of MBRSHP-HOUSEHOLD
                                           TO FIN435-I-CLUB-CD.
           MOVE SPACES                     TO FIN435-I-PROD-PREFIX.
           MOVE MBRSHP-N                   of MBRSHP-HOUSEHOLD
                                           TO FIN435-I-PRODUCT-CD.
           MOVE SPACES                     TO FIN435-I-DATE-CCYYMMDD.

           CALL WS-FN435BT          USING  FN435BT-COMMAREA

           IF FIN435-RETURN-CODE = '000'
              PERFORM 2005-CHECK-PIT-DETAIL THRU 2005-X
           ELSE
              IF FIN435-RETURN-CODE = '210'
                 SET MEMBER-HAS-PIT-NO  TO TRUE
              ELSE
               DISPLAY '********************************************'
               DISPLAY '** ERROR CALLING PIT     USING CLUB/MBR:  **'
               DISPLAY '** '
                       FK-ACL-CLUB-C of MBRSHP-HOUSEHOLD
                       '/'
                       MBRSHP-N of MBRSHP-HOUSEHOLD
               DISPLAY '********************************************'
               DISPLAY '**FIN435-RETURN-CODE: ' FIN435-RETURN-CODE
               DISPLAY '**FIN435-ERROR-MSG: ' FIN435-RETURN-MSG
                 MOVE FK-ACL-CLUB-C         OF MBRSHP-HOUSEHOLD
                                            TO MD900-KEY-DATA-LN (1)
                 MOVE MBRSHP-N              OF MBRSHP-HOUSEHOLD
                                            TO MD900-KEY-DATA-LN (2)
                 MOVE FIN435-RETURN-CODE    TO MD900-KEY-DATA-LN (3)
                 STRING 'BAD RETURN CODE FROM FN435BT: '
                  FIN435-RETURN-CODE    DELIMITED BY SIZE
                                    INTO MD900-ERROR-MSG-1
                 MOVE FIN435-RETURN-MSG
                                   TO MD900-ERROR-MSG-2
                 MOVE 'FN435BT'
                                   TO MD900-PGM-NAME
                                      MD900-TABLE-CURSOR-NAME
                 MOVE '2001-CHECK-PIT'
                                    TO MD900-PARAGRAPH-NAME
                 PERFORM 9992-CLOSE-CURSOR THRU 9992-X
                 PERFORM 9999-ABEND THRU 9999-X
              END-IF
           END-IF.


       2001-X.  EXIT.

       2005-CHECK-PIT-DETAIL.

           PERFORM VARYING WS-PIT-SUB FROM +1 BY +1
                UNTIL  WS-PIT-SUB > FIN435-O-NBR-OF-PAYMENTS
                   OR  MAX-PIT-SUB
             IF FIN435-O-PAYMNT-STATUS(WS-PIT-SUB) = 'PIT '
                ADD FIN435-O-PAYMNT-AMOUNT(WS-PIT-SUB) TO WS-PIT-TOTAL
             END-IF
           END-PERFORM.

           IF WS-PIT-TOTAL > 0
              SET MEMBER-HAS-PIT-YES TO TRUE
           ELSE
              SET MEMBER-HAS-PIT-NO  TO TRUE
           END-IF.


       2005-X.  EXIT.

       2010-CALL-MC501BCD.

           INITIALIZE  MC501-INPUT-OUTPUT.                                 CL**1
                                                                           CL**1
           MOVE MD021-CUST-ID        TO MC501-I-CUSTOMER-NUMBER.           CL**1
           MOVE '01'                 TO MC501-I-LOB-CODE.                  CL**1
           MOVE 'MD021EX '           TO MC501-I-CALLING-PROGRAM.           CL**1
           MOVE '00'                 TO MC501-I-ROLE-CODE.                 CL**1
                                                                           CL**1
           MOVE 'Y'                  TO MC501-I-REQ-ALL-CUST-DATA.         CL**1
           MOVE 'Y'                  TO MC501-I-REQ-ALL-CUST-PHONE.        CL**1
                                                                           CL**1
           CALL WS-MC501BCD    USING  MC501-INPUT-OUTPUT.                  CL**1

           EVALUATE MC501-O-RETURN-CODE
             WHEN ZERO
      **** MEM-479718 START
                MOVE MC501-O-FIRST-NAME  TO MD021-FIRST-NAME
      **** MEM-479718 END
                  IF MC501-O-EMAIL-VALID-C = '1'
                     MOVE MC501-O-EMAIL      TO MD021-EMAIL
                                                MD021-SUBSCRIBER-KEY
                  END-IF
                  PERFORM VARYING PHONE-IND FROM 1 BY 1
                    UNTIL PHONE-IND > MC501-O-NUM-OF-PHONES
                     IF MC501-O-PH-ROLE (PHONE-IND) = 'CL' AND
                        MC501-O-PH-VALID-CODE (PHONE-IND) = '1' AND
                        MC501-O-PH-EXP-DATE (PHONE-IND) = '99991231'
                        MOVE '1'
                          TO MD021-CELL-PHONE (1:1)
                        MOVE MC501-O-PH-AREA-CODE (PHONE-IND)
                          TO MD021-CELL-PHONE (2:3)
                        MOVE MC501-O-PH-PREFIX (PHONE-IND)
                          TO MD021-CELL-PHONE (5:3)
                        MOVE MC501-O-PH-SUFFIX (PHONE-IND)
                          TO MD021-CELL-PHONE (8:4)
                     END-IF
                  END-PERFORM
             WHEN OTHER
             DISPLAY '********************************************'
             DISPLAY '** ERROR CALLING MC501   USING CLUB/MBR:  **'
             DISPLAY '** '
                     FK-ACL-CLUB-C OF MBRSHP-HOUSEHOLD
                     '/'
                     MBRSHP-N OF MBRSHP-HOUSEHOLD
             DISPLAY '********************************************'
             DISPLAY '**MC501-O-RETURN-CODE: ' MC501-O-RETURN-CODE
             DISPLAY '**MC501-O-RETURN-MESSAGE: ' MC501-O-RETURN-MESSAGE
                 MOVE FK-ACL-CLUB-C         OF MBRSHP-HOUSEHOLD
                                            TO MD900-KEY-DATA-LN (1)
                 MOVE MBRSHP-N              OF MBRSHP-HOUSEHOLD
                                            TO MD900-KEY-DATA-LN (2)
                 MOVE MC501-O-RETURN-CODE   TO MD900-KEY-DATA-LN (3)
                 STRING 'BAD RETURN CODE FROM MC501BCD: '
                  MC501-O-RETURN-CODE   DELIMITED BY SIZE
                                    INTO MD900-ERROR-MSG-1
                 MOVE MC501-O-RETURN-MESSAGE
                                   TO MD900-ERROR-MSG-2
                 MOVE 'MC501BCD'
                                   TO MD900-PGM-NAME
                                      MD900-TABLE-CURSOR-NAME
                 MOVE '2010-CALL-MC501BCD'
                                    TO MD900-PARAGRAPH-NAME
                 PERFORM 9992-CLOSE-CURSOR THRU 9992-X
                 PERFORM 9999-ABEND THRU 9999-X
           END-EVALUATE
           .

      *** NOTE: THERE IS NO RETURN CODE OR ERROR INFO COMING BACK FROM
      *** MD999CK. IT EITHER ABENDS OR RETURNS THE DATA.

           CALL WS-MD999CK           USING MD999-LINK-DATA.

       2015-X.  EXIT.

              WRITE OUTREC FROM WS-OUTREC

              .
       2020-X.  EXIT.
